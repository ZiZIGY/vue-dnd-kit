const el = useTemplateRef<HTMLElement>('el');
const { isDragging } = makeDraggable(el, { id: 'card', type: 'card', data: { title } });

déplaçables : [`makeSortableItem`](/docs/api/make-sortable-item) accepte les mêmes options, et
d'autres encore.
